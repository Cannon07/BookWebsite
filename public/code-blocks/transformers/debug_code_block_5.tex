class Encoder(L: $\mathbb{N}$, M: $\mathbb{N}$, H: $\mathbb{N}$, F: $\mathbb{N}$, dropout: $\mathbb{R}$):
    layers = [
        EncoderLayer(M, H, F, dropout) for _ in L]

    def $\\lambda$(src: $\mathbb{R}$[L, D]):
        src += PositionEncoding(L, D)
        for layer in layers:
          src = layer(src)
        src
