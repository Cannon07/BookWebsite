def occurs(v: Var, term: Term, $\\Gamma$: Optional[Name $\to$ Term]) $\to$ Bool:
  if v == term: True
  elif term : Var and term.name in $\\Gamma$: occurs(v, $\\Gamma$[term.name], $\\Gamma$)
  elif term : App:
    any(occurs(v, arg, $\\Gamma$) for arg in term.args)
  else: False
\end{lstlisting}
