def unify_all_equations(ast: AST):
  """Unifies all type equations in the sequence eqs.    """
  eqs = generate_equations(ast)
  $\\Gamma$ = {}
  for eq in eqs:
    $\\Gamma$ = unify(eq.left, eq.right, $\\Gamma$)
    if $\\Gamma$ is None:
      break
  $\\Gamma$
