def R(r: $\mathbb{R}$, rmax: $\mathbb{R}$, params: $\mathbb{R}^3$) -> $\mathbb{R}$
  if r > rmax:
    return 0

  c, pow = params[1:2]
  res = r ≈ 0 ? 0 : c / r^pow

  c = params[3]
  res += r ≈ 0 ? c : 0
  return res
