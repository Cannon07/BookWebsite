def adaptive_euler(f: $\mathbb{R} \\to \\mathbb{R}$, $(t_0, t_e)$: $\mathbb{R}^2$, tol: $\mathbb{R}$) $\to \\mathbb{R}[]$:
  t_adp, y, dt: $\mathbb{R}[]$
  t_adp[0] = $t_0$, i = 0
  y[0] = 0
  dt[0] = sqrt(2*tol/$\nabla$f(t_adp[0]))
  while (t_adp[i] < $t_e$):
    y[i+1] = y[i] + f(t_adp[i])*dt[i]
    dt[i+1] = sqrt(2*tol/$\nabla$f(t_adp[i]))
    t_adp[i+1] = t_adp[i] + dt[i+1]
    i = i + 1
  y
