def naive_gaussian_elimination(A : $\mathbb{R}^{N\\times N}$, B: $\mathbb{R}^N$):
  x = zeros(N)
  Aug = [A B]
  for i = 0:N-1, j = i+1:N
    ratio = Aug[j,i]/Aug[i,i]
    Aug[j,i:N+1] = Aug[j,i:N+1]-ratio*Aug[i,i:N+1]
  x[N-1] = Aug[N-1,N]/Aug[N-1,N-1]
  for i = N-2:-1:0
    x[i] = (Aug[i,N+1]-Aug[i,i+1:N]*x[i+1:N])/Aug[i,i]
  x
