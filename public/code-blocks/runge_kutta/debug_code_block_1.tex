def initialize_grid(N: $\mathbb{N}$) $\to \\mathbb{R}^{N \\times N}$:
  C = zeros(N, N)
  for i j:
    temp = randi(2)
    if (temp==1):
      C[j,i] = -1
    else 
      C[j,i] = 1
