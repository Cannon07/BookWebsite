def secant_method(f: $\mathbb{R} \\to \\mathbb{R}$, now: $\mathbb{R}$ = 120, prev: $\mathbb{R}$ = 100): $\mathbb{R}$                                            
  while True: 
    f_prime = (f(prev)-f(now))/(prev-now)        
    new = now-f(now)/f_prime                       
    # Convergence criterion 
    if abs((new-now)/new) < 10**(-8):           
      break 
    prev = now
    now = new
  return new
