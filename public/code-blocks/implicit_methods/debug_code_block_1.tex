def root2d(y, y1, y2, $\\mu$, h):
 y1dash = y[0], y2dash = y[1]
 F[0] = -y1dash + y1 + (y2dash)*h
 F[1] = -y2dash + y2 + (($\\mu$*(1-y1dash$^2$)*y2dash - y1dash))*h
 return F
