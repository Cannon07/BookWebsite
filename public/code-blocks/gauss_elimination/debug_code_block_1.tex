def pivoting(A : $\mathbb{R}^{N \\times N}$, B : $\mathbb{R}^N$):
  x = zeros(N,1)
  Aug = [A B]
  for k = 0:N-2
    [big,index_change] = max(abs(Aug[k:N,k]))
    index = index_change + k - 1
    if index != k
      Aug[k,index,:] = Aug[index,k,:]
    for index = k+1:N
      ratio = Aug[index,k]/Aug[k,k]
      Aug[index,k:B+1] = Aug[index,k:B+1] - ratio*Aug[k,k:B+1]
