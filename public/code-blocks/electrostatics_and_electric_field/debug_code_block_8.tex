def E($n$: $\mathbb{N}$, [$q_i$]: $\mathbb{R}^n$, [$x_i$]: $\mathbb{R}^{3n}$, x: $\mathbb{R}^3$) -> $\mathbb{R}^3$:
  return sum([E($\epsilon_0$, qi, xi, x) 
              for (qi, xi) in zip([$q_i$], [$x_i$])])
