def f(x: $\mathbb{R}$) -> $\mathbb{R}$:
  return exp(x)-2*x                                        
xl = -10                                                   
xu = 10 
eps = 1e-6                                                 
phi = (-1+sqrt(5))/2
x1 = xl + phi*(xu-xl)                                      
x2 = xu - phi*(xu-xl)                                      
fx1 = f(x1)                                                
fx2 = f(x2)
while True:
  if fx1 < fx2:
    xl = x2                                                
    x2 = x1                                                
    fx2 = fx1                                              
    x1 = xl + phi*(xu-xl)                                  
    fx1 = f(x1)                                            
  else:
    xu = x1
    x1 = x2 
    fx1 = fx2
    x2 = xu - phi*(xu-xl)                                  
    fx2 = f(x2)                                            
  if abs(xu-xl) < eps:                                     
    break 
print('Extremum bracket')                                  
print([xl, xu])
