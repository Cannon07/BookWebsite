class DiscreteGroupConvolution(G: Group, w: $\mathbb{R}[|G|]$):
  def $\\lambda$(X: $\mathbb{R}[|G|]$) -> $\mathbb{R}[|G|]$:
    X$'$ = zeros(|G|)
    for g h:
      X$'$[h] += w[g]*X[$g^{-1}$*h]
    return X$'$
