def metropolis_hastings($\\theta: \\mathbb{R}$, $q: \\mathbb{R} \\to \\mathbb{R}$, $\\beta: \\mathbb{R}$, N : $\mathbb{N}$):
  $E_\\theta$ = E($\\theta$)
  for i < N:
    $\\theta^*$ = $q(\\theta)$
    $E_{\\theta^*}$ = E($\\theta^*$)
    $\alpha$ = exp($\\beta(E_\\theta - E_{\\theta^*}$)
    if $\alpha$ >= 1:
      $\\theta$ = $\\theta^*$
    else:
      A = uniform(0, 1)
      if A < $\alpha$:
        $\\theta$ = $\\theta^*$
