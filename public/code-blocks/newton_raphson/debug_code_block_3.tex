def newton_raphson_opt(f: $\mathbb{R} \\to \\mathbb{R}$, start: $\mathbb{R}$, tol:$\mathbb{R}$ = 1e-8): 
  old = start
  while True: 
    new = old-grad(f)(old)/grad(grad(f))(old)
    if abs((new-old)/new) < tol:
      break
    old = new
  return new
