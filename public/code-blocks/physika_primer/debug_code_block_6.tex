def matmul(A: $\mathbb{R}$[n, m], B: $\mathbb{R}$[m, o]) $\to$ $\mathbb{R}$[n, o]:
  C: $\mathbb{R}$[n, o] = 0
  for i j k:
    C[i, j] += A[i,k]*B[k, j]
  C
