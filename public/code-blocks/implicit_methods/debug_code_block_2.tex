def implicit_euler_method($\\mu$: $\mathbb{R}$, t$_f$: $\mathbb{R}$, dt: $\mathbb{R}$):
  t = 0:dt:t$_f$
  n = len(t)
  y1i = zeros(n,1), y2i = zeros(n,1)
  y1i[0] = 1, y2i[0] = 1
  for i in 1 .. n-1:
    solution = fsolve($\\lambda$ (y) root2d(y,y1i[i],y2i[i],$\\mu$,dt),  [y1i[i], y2i[i]])
    y1i[i+1] = solution[0]
    y2i[i+1] = solution[1]
