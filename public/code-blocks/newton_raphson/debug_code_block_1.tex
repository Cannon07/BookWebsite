def newton_raphson(f: $\mathbb{R} \\to \\mathbb{R}$, start: $\mathbb{R}$ = 0): $\mathbb{R}$  
  old = start
  while True:
    new = old - f(old)/grad(f)(old)                 
    # Convergence criterion                              
    if abs((new-old)/new) < 10**(-8):           
      break
    old = new
  return new
