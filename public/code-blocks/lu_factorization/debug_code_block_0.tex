def lu_factorization(A: $\mathbb{R}^{N \\times N}$):
  L = eye(N)
  U = A
  for i = 0:N-1
    for j = i+1:N
      ratio = U[j,i]/U[i,i]
      U[j,i:N] = U[j,i:N]-ratio*U[i,i:N]
      L[j,i] = ratio
  i=U\\(L\\b)
