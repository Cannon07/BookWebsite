def rbf($X: \\mathbb{R}^{M \\times N}$, $\\sigma$) $\to$ $\mathbb{R}^{M \\times M}$:
  K: $\mathbb{R}^{M \\times M}$
  for i j
    K[i,j] = exp(-$\\|X[i]-X[j]\\|^2$/($2\\sigma^2$))
  return K
